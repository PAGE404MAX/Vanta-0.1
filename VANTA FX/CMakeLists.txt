cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(VantaFX VERSION 0.1.0 LANGUAGES C CXX)
 set(CMAKE_CXX_STANDARD 17)
 set(CMAKE_CXX_STANDARD_REQUIRED ON)
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(VantaFX VERSION 0.1.0 COMPANY_NAME Vanta BUNDLE_ID com.vanta.fx PLUGIN_MANUFACTURER_CODE Vnta PLUGIN_CODE Vfxa FORMATS VST3 PRODUCT_NAME "VANTA FX" VST3_CATEGORIES Fx Reverb Delay Modulation IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(VantaFX)
target_sources(VantaFX PRIVATE Source/Plugin.cpp)
target_compile_definitions(VantaFX PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(VantaFX PRIVATE juce::juce_audio_utils juce::juce_dsp)
option(VFX_BUILD_TESTS "Build FX engine tests" ON)
if(VFX_BUILD_TESTS)
 enable_testing()
 add_executable(VantaFXEngineTests tests/EngineTests.cpp)
 target_include_directories(VantaFXEngineTests PRIVATE Source)
 add_test(NAME VantaFXEngineTests COMMAND VantaFXEngineTests)
 juce_add_console_app(VantaFXProcessorTests PRODUCT_NAME "VantaFXProcessorTests")
 juce_generate_juce_header(VantaFXProcessorTests)
 target_sources(VantaFXProcessorTests PRIVATE tests/ProcessorTests.cpp)
 target_compile_definitions(VantaFXProcessorTests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(VantaFXProcessorTests PRIVATE juce::juce_audio_utils juce::juce_dsp)
 add_test(NAME VantaFXProcessorTests COMMAND VantaFXProcessorTests)
endif()
